```ts
// Edge function: estimate-prezzo-mercato
// Stima il prezzo unitario di mercato per una lavorazione, usando il prezzario
// regionale (prezzi_unitari globali) e cicli globali come riferimento.
// verify_jwt = true.

import { createClient } from "npm:@supabase/supabase-js@2";

const ANTHROPIC_API_KEY = Deno.env.get("ANTHROPIC_API_KEY")!;
const MODEL = Deno.env.get("ANTHROPIC_MODEL") ?? "claude-opus-4-7";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

const SCHEMA = {
  type: "object",
  additionalProperties: false,
  properties: {
    prezzo_mercato: { type: "number" },
    fonte: { type: "string" },
    note: { type: "string" },
  },
  required: ["prezzo_mercato", "fonte"],
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });
  try {
    const authHeader = req.headers.get("Authorization") ?? "";
    const supabase = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_ANON_KEY")!,
      { global: { headers: { Authorization: authHeader } } },
    );

    // lavorazione (titolo), unita, comune (per costo locale)
    const { lavorazione, unita, comune } = await req.json();
    if (!lavorazione) return json({ error: "lavorazione richiesta" }, 400);

    // Prezzario regionale = prezzi_unitari globali (org_id is null)
    const { data: prezzi } = await supabase
      .from("prezzi_unitari")
      .select("voce, unita, prezzo, categoria")
      .is("org_id", null);

    const listino = (prezzi ?? [])
      .map((p) => `- ${p.voce} (${p.unita}): ${p.prezzo} €`)
      .join("\n");

    const body = {
      model: MODEL,
      max_tokens: 600,
      system: [
        {
          type: "text",
          text:
            "Sei un computista edile. Stima il PREZZO UNITARIO DI MERCATO per la lavorazione " +
            "indicata, riferito a ristrutturazione di facciate in Italia" +
            (comune ? ` (zona: ${comune})` : "") +
            ". Usa come riferimento questo prezzario regionale; se la voce non c'è, " +
            "deduci un valore realistico per analogia. Restituisci un singolo numero in euro " +
            "coerente con l'unità di misura richiesta.\n\nPREZZARIO:\n" + listino,
          cache_control: { type: "ephemeral" },
        },
      ],
      messages: [
        {
          role: "user",
          content:
            `Lavorazione: ${lavorazione}\nUnità: ${unita ?? "corpo"}\n` +
            `Stima il prezzo unitario di mercato e indica la fonte/criterio.`,
        },
      ],
      output_config: { effort: "low", format: { type: "json_schema", schema: SCHEMA } },
    };

    const aiRes = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "x-api-key": ANTHROPIC_API_KEY,
        "anthropic-version": "2023-06-01",
        "content-type": "application/json",
      },
      body: JSON.stringify(body),
    });
    if (!aiRes.ok) return json({ error: "Errore AI", detail: await aiRes.text() }, 502);
    const aiData = await aiRes.json();
    const textBlock = (aiData.content ?? []).find((b: { type: string }) => b.type === "text");
    return json({ ok: true, stima: JSON.parse(textBlock.text) });
  } catch (e) {
    return json({ error: String(e) }, 500);
  }
});

function json(obj: unknown, status = 200) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { ...corsHeaders, "content-type": "application/json" },
  });
}
```

```ts
export async function addCicloPitturazione(
  preventivoId: string,
  superficieMq: number | null,
) {
  const { org } = await requireContext();
  const supabase = await createClient();

  // Cerca un ciclo "pitturazione" in libreria (org poi globale)
  const { data: cicli } = await supabase
    .from("cicli_lavorazione")
    .select("*")
    .ilike("titolo", "%pitturazione%")
    .order("org_id", { ascending: false, nullsFirst: false })
    .limit(1);
  const ciclo = cicli?.[0];

  const { count } = await supabase
    .from("preventivo_interventi")
    .select("id", { count: "exact", head: true })
    .eq("preventivo_id", preventivoId);

  await supabase.from("preventivo_interventi").insert({
    org_id: org.id,
    preventivo_id: preventivoId,
    ordine: count ?? 0,
    titolo: ciclo?.titolo ?? "Pitturazione totale della superficie",
    descrizione:
      ciclo?.descrizione_fasi ??
      "Ciclo completo di pitturazione della superficie: preparazione del supporto, applicazione fissativo e finitura.",
    unita: ciclo?.unita_default ?? "mq",
    quantita: superficieMq ?? 1,
    prezzo_unitario: ciclo?.prezzo_default ?? 0,
    ciclo_id: ciclo?.id ?? null,
```

```ts
            { type: "text", text: "Annotazioni utente (JSON):\n" + annotazioni + "\n\nGenera gli interventi." },
          ],
        },
      ],
      output_config: { effort: "medium", format: { type: "json_schema", schema: SCHEMA } },
    };

    const aiRes = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "x-api-key": ANTHROPIC_API_KEY,
        "anthropic-version": "2023-06-01",
        "content-type": "application/json",
      },
      body: JSON.stringify(body),
    });
    if (!aiRes.ok) return json({ error: "Errore AI", detail: await aiRes.text() }, 502);
    const aiData = await aiRes.json();
    const textBlock = (aiData.content ?? []).find((b: { type: string }) => b.type === "text");
    return json({ ok: true, risultato: JSON.parse(textBlock.text) });
  } catch (e) {
    return json({ error: String(e) }, 500);
  }
});

function json(obj: unknown, status = 200) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { ...corsHeaders, "content-type": "application/json" },
  });
}
```

```tsx
"use client";

import { useRef, useState } from "react";

import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/field";

export type Punto = { x: number; y: number; numero: number; lavorazione: string };

export function FotoAutopsia({
  imageUrl,
  initialPunti = [],
  onAvvia,
  onSave,
}: {
  imageUrl: string;
  initialPunti?: Punto[];
  onAvvia: (punti: Punto[], legenda: Record<string, string>) => Promise<void>;
  onSave?: (punti: Punto[], legenda: Record<string, string>) => Promise<void>;
}) {
  const [punti, setPunti] = useState<Punto[]>(initialPunti);
  const [next, setNext] = useState(1);
  const [loading, setLoading] = useState(false);
  const imgRef = useRef<HTMLImageElement>(null);

  function addPunto(e: React.MouseEvent<HTMLDivElement>) {
    const rect = e.currentTarget.getBoundingClientRect();
    const x = (e.clientX - rect.left) / rect.width;
    const y = (e.clientY - rect.top) / rect.height;
    const numero = next > 9 ? 9 : next;
    setPunti((p) => [...p, { x, y, numero, lavorazione: "" }]);
    setNext((n) => Math.min(n + 1, 9));
  }

  function setLavorazione(idx: number, value: string) {
    setPunti((p) => p.map((pt, i) => (i === idx ? { ...pt, lavorazione: value } : pt)));
  }

  function removePunto(idx: number) {
    setPunti((p) => p.filter((_, i) => i !== idx));
  }

  function legendaFrom(pts: Punto[]): Record<string, string> {
    const leg: Record<string, string> = {};
    for (const pt of pts) if (pt.lavorazione) leg[String(pt.numero)] = pt.lavorazione;
    return leg;
  }

  return (
    <div className="space-y-4">
      <div
        onClick={addPunto}
        className="relative cursor-crosshair overflow-hidden rounded-md border border-hairline"
      >
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img ref={imgRef} src={imageUrl} alt="Facciata" className="w-full select-none" />
        {punti.map((pt, i) => (
          <span
            key={i}
            className="absolute flex h-7 w-7 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border-2 border-canvas bg-ink text-xs font-semibold text-canvas"
            style={{ left: `${pt.x * 100}%`, top: `${pt.y * 100}%` }}
          >
            {pt.numero}
          </span>
        ))}
      </div>

      <ol className="space-y-2">
        {punti.map((pt, i) => (
          <li key={i} className="flex items-center gap-2">
            <span className="flex h-6 w-6 items-center justify-center rounded-full bg-ink text-xs text-canvas">
              {pt.numero}
            </span>
            <Input
              value={pt.lavorazione}
              placeholder="Tipo di lavorazione / ripristino"
              onChange={(e) => setLavorazione(i, e.target.value)}
            />
            <button type="button" onClick={() => removePunto(i)} className="text-xs text-stone underline">
              Rimuovi
            </button>
          </li>
        ))}
        {punti.length === 0 ? (
          <li className="text-sm text-stone">Clicca sulla foto per posizionare i punti.</li>
        ) : null}
      </ol>

      <div className="flex gap-2">
        {onSave ? (
          <Button
            variant="secondary"
            onClick={() => onSave(punti, legendaFrom(punti))}
          >
            Salva annotazioni
          </Button>
        ) : null}
        <Button
          disabled={loading || punti.length === 0}
          onClick={async () => {
            setLoading(true);
            try {
              await onAvvia(punti, legendaFrom(punti));
            } finally {
              setLoading(false);
            }
          }}
        >
          {loading ? "Generazione…" : "Avvia"}
        </Button>
      </div>
    </div>
  );
}
```
